     ```go
     type ChatRequest struct {
         Model    string        `json:"model"`
         Messages []ChatMessage `json:"messages"`
         Stream   bool          `json:"stream"`
     }

     // ChatResponse is the non-streamed chat result.
     type ChatResponse struct {
         Message ChatMessage `json:"message"`
     }

     // ExtractRequest is the payload for Extract.
     type ExtractRequest struct {
         // DocumentPath is the absolute path inside the shared uploads volume.
         DocumentPath string `json:"document_path"`
     }

     // ExtractResponse is the extraction result (Phase 3 defines full shape).
     type ExtractResponse struct {
         Status string `json:"status"`
     }

     // EngineClient is the seam. gRPC-ready: a future grpcClient implements
     // the same interface; callers never see the transport.
     type EngineClient interface {
         // Chat sends a JSON request and waits for the full JSON response.
         // Budget: seconds (default 30s).
         Chat(ctx context.Context, req ChatRequest) (*ChatResponse, error)

         // Budget: seconds (default 10s).
     ```

     ```go
     package engine

     import (
         "bufio"
         "bytes"
         "context"
         "fmt"
         "io"
         "strings"
     )

     // scanSSEEvents reads an SSE stream from r and calls fn for every event.
     // Splits on blank lines (\n\n or \r\n\r\n), tolerates comment lines
     // (heartbeats ": ping"), joins multi-line data fields per SSE spec,
     // and stops on ctx cancellation.
     func scanSSEEvents(ctx context.Context, r io.Reader, fn func(EngineEvent) error) error {
         scanner := bufio.NewScanner(r)
         scanner.Buffer(make([]byte, 0, 64*1024), 1024*1024) // >64KB default; 1MB cap
         scanner.Split(splitSSEEvent)

         for scanner.Scan() {
             if err := ctx.Err(); err != nil {
                 return err
             }
             block := scanner.Bytes()
             if len(bytes.TrimSpace(block)) == 0 {
                 continue // heartbeat/keep-alive comment block
             }
             evt := parseSSEBlock(block)
             if evt == nil {
                 continue // comment-only block (": ping")
             }
             if err := fn(*evt); err != nil {
                 return err
             }
         }
         return scanner.Err()
     }

     // splitSSEEvent advances to the next blank line (event boundary).
     func splitSSEEvent(data []byte, atEOF bool) (advance int, token []byte, err error) {
         if atEOF && len(data) == 0 {
             return 0, nil, nil
         }
         if i := bytes.Index(data, []byte("\n\n")); i >= 0 {
             return i + 2, data[:i], nil
         }
         if i := bytes.Index(data, []byte("\r\n\r\n")); i >= 0 {
             return i + 4, data[:i], nil
         }
         if atEOF {
             return len(data), data, nil
         }
         return 0, nil, nil // need more data
     }

     // parseSSEBlock parses "event:" and "data:" fields from one event block.
     // Returns nil for comment-only blocks.
     func parseSSEBlock(block []byte) *EngineEvent {
         var evtType, data strings.Builder
         for _, line := range strings.Split(string(block), "\n") {
             line = strings.TrimSuffix(line, "\r")
             switch {
             case strings.HasPrefix(line, ":"):
                 continue // comment
             case strings.HasPrefix(line, "event:"):
                 evtType.WriteString(strings.TrimSpace(strings.TrimPrefix(line, "event:")))
             case strings.HasPrefix(line, "data:"):
                 if data.Len() > 0 {
                     data.WriteByte('\n') // multi-line data joins with \n
                 }
                 data.WriteString(strings.TrimSpace(strings.TrimPrefix(line, "data:")))
             }
         }
         if evtType.Len() == 0 && data.Len() == 0 {
             return nil
         }
         return &EngineEvent{Type: evtType.String(), Data: []byte(data.String())}
     }
     ```

  2. Create `client.go` — implement the contract:
     ```go
     package engine

     import (
         "bytes"
         "context"
         "encoding/json"
         "fmt"
         "io"
         "net/http"
         "time"

         "github.com/google/uuid"

         "github.com/playbits/schoolcare-v2/internal/middleware"
     )

     const (
         chatTimeout   = 30 * time.Second
         extractTimeout = 5 * time.Minute
         healthTimeout = 10 * time.Second
         // ChatStream has NO timeout constant — context-bound by design (FND-03).
     )

     // httpClient implements EngineClient over REST/JSON + SSE.
     type httpClient struct {
         baseURL string
         token   string
         client  *http.Client // no global timeout; per-endpoint via request ctx
     }

     // NewClient creates an EngineClient for the given base URL and service token.
     func NewClient(baseURL, token string) EngineClient {
         return &httpClient{
             baseURL: baseURL,
             token:   token,
             client:  &http.Client{},
         }
     }

     func (c *httpClient) Chat(ctx context.Context, req ChatRequest) (*ChatResponse, error) {
         ctx, cancel := context.WithTimeout(ctx, chatTimeout)
         defer cancel()
         body, err := json.Marshal(req)
         if err != nil {
             return nil, fmt.Errorf("engine chat marshal: %w", err)
         }
         hreq, err := c.newRequest(ctx, http.MethodPost, "/v1/chat", body)
         if err != nil {
             return nil, err
         }
         resp, err := c.client.Do(hreq)
         if err != nil {
             return nil, fmt.Errorf("engine chat: %w", err)
         }
         defer resp.Body.Close()
         if resp.StatusCode != http.StatusOK {
             return nil, fmt.Errorf("engine chat: unexpected status %d", resp.StatusCode)
         }
         var out ChatResponse
         if err := json.NewDecoder(resp.Body).Decode(&out); err != nil {
             return nil, fmt.Errorf("engine chat decode: %w", err)
         }
         return &out, nil
     }

     func (c *httpClient) ChatStream(ctx context.Context, req ChatRequest, cb StreamCallback) error {
         // NOTE: no WithTimeout here — stream is context-bound (FND-03).
         req.Stream = true
         body, err := json.Marshal(req)
         if err != nil {
             return fmt.Errorf("engine chatstream marshal: %w", err)
         }
         hreq, err := c.newRequest(ctx, http.MethodPost, "/v1/chat/stream", body)
         if err != nil {
             return err
         }
         resp, err := c.client.Do(hreq)
         if err != nil {
             return fmt.Errorf("engine chatstream: %w", err)
         }
         defer resp.Body.Close()
         if resp.StatusCode != http.StatusOK {
             return fmt.Errorf("engine chatstream: unexpected status %d", resp.StatusCode)
         }
         return scanSSEEvents(ctx, resp.Body, func(evt EngineEvent) error {
             if err := cb(evt); err != nil {
                 return fmt.Errorf("engine chatstream callback: %w", err)
             }
             return nil
         })
     }

     func (c *httpClient) Extract(ctx context.Context, req ExtractRequest) (*ExtractResponse, error) {
         ctx, cancel := context.WithTimeout(ctx, extractTimeout)
         defer cancel()
         body, err := json.Marshal(req)
         if err != nil {
             return nil, fmt.Errorf("engine extract marshal: %w", err)
         }
         hreq, err := c.newRequest(ctx, http.MethodPost, "/v1/extract", body)
         if err != nil {
             return nil, err
         }
         resp, err := c.client.Do(hreq)
         if err != nil {
             return nil, fmt.Errorf("engine extract: %w", err)
         }
         defer resp.Body.Close()
         if resp.StatusCode != http.StatusOK {
             return nil, fmt.Errorf("engine extract: unexpected status %d", resp.StatusCode)
         }
         var out ExtractResponse
         if err := json.NewDecoder(resp.Body).Decode(&out); err != nil {
             return nil, fmt.Errorf("engine extract decode: %w", err)
         }
         return &out, nil
     }

     func (c *httpClient) Health(ctx context.Context) error {
         ctx, cancel := context.WithTimeout(ctx, healthTimeout)
         defer cancel()
         hreq, err := c.newRequest(ctx, http.MethodGet, "/v1/health", nil)
         if err != nil {
             return err
         }
         resp, err := c.client.Do(hreq)
         if err != nil {
             return fmt.Errorf("engine health: %w", err)
         }
         defer resp.Body.Close()
         if resp.StatusCode != http.StatusOK {
             return fmt.Errorf("engine health: unexpected status %d", resp.StatusCode)
         }
         return nil
     }

     // newRequest builds a request with service-token auth and correlation headers.
     func (c *httpClient) newRequest(ctx context.Context, method, path string, body []byte) (*http.Request, error) {
         var reader io.Reader
         if body != nil {
             reader = bytes.NewReader(body)
         }
         hreq, err := http.NewRequestWithContext(ctx, method, c.baseURL+path, reader)
         if err != nil {
             return nil, fmt.Errorf("engine request: %w", err)
         }
         hreq.Header.Set("Content-Type", "application/json")
         hreq.Header.Set(EngineTokenHeader, c.token) // NEVER in URL (Rule FND-04/SC4)
         reqID := middleware.GetRequestIDFromCtx(ctx)
         if reqID == "" {
             reqID = uuid.NewString()
         }
         hreq.Header.Set(RequestIDHeader, reqID)
         return hreq, nil
     }
     ```
