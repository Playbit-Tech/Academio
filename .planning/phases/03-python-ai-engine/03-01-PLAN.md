```proto
syntax = "proto3";

package aiengine.v1;

// AiEngine is the Python AI engine contract seam (PYE-04a).
// v1 transports over REST/JSON + SSE; this file is the single source of
// truth for request/response semantics. Transport may become gRPC later.
service AiEngine {
  // Chat maps to POST /v1/chat (non-streaming completion).
  rpc Chat(ChatRequest) returns (ChatResponse);
  // ChatStream maps to POST /v1/chat/stream (server-streaming SSE).
  rpc ChatStream(ChatRequest) returns (stream EngineEvent);
  // Embed maps to POST /v1/embed (canonical 1536-dim embeddings).
  rpc Embed(EmbedRequest) returns (EmbedResponse);
  // Extract maps to POST /v1/extract (document text intelligence only).
  rpc Extract(ExtractRequest) returns (ExtractResponse);
  // IngestDocument maps to POST /v1/documents (extract+chunk+embed+store).
  rpc IngestDocument(IngestDocumentRequest) returns (IngestDocumentResponse);
  // Search maps to POST /v1/search (tenant-aware hybrid RAG).
  rpc Search(SearchRequest) returns (SearchResponse);
}

message ChatMessage {
  string role = 1;    // "user" | "assistant" | "system"
  string content = 2;
}

message ChatRequest {
  string model = 1;             // "provider:model" composite (D-03), e.g. "anthropic:claude-3-5-sonnet-latest"
  repeated ChatMessage messages = 2;
  bool stream = 3;              // ignored on ChatStream; set true by Go client
}

message Usage {
  string provider = 1;
  string model = 2;
  int64 input_tokens = 3;
  int64 output_tokens = 4;
  double cost = 5;              // USD, computed by cost.py (D-03)
}

message ChatResponse {
  ChatMessage message = 1;
  Usage usage = 2;              // normalized usage on EVERY response (ROADMAP criterion 2)
}

message EngineEvent {
  string type = 1;              // "delta" | "citation" | "usage" | "error" | "done" (D-02)
  bytes data = 2;               // JSON object; compact single-line in SSE transport
}

message EmbedRequest {
  repeated string texts = 1;    // batch <= 128 per call (D-05)
}

message Embedding {
  repeated float values = 1;    // must be exactly 1536 floats (AI_EMBEDDING_DIM)
}

message EmbedResponse {
  string model = 1;             // "text-embedding-3-small" (canonical)
  int32 dimension = 2;          // 1536
  repeated Embedding embeddings = 3;
}

message ExtractRequest {
  string document_path = 1;     // absolute path inside shared uploads volume
}

message ExtractResponse {
  string status = 1;            // "success" | "error"
  int32 pages = 2;
  int32 ocr_pages = 3;          // pages that went through Tesseract OCR
  int64 chars = 4;              // total extracted characters (quality gate)
}

message IngestDocumentRequest {
  string document_path = 1;     // absolute path inside shared uploads volume
  string schema_name = 2;       // "school_1" — validated ^school_[0-9]+$ + existence (D-09/D-07)
  string collection = 3;        // optional corpus name; defaults "default"
}

message IngestDocumentResponse {
  string status = 1;            // "success" | "error"
  string document_id = 2;       // uuid used for ai_vectors.document_id (unique with chunk_index)
  int32 chunks = 3;
  int32 pages = 4;
}

message Filter {
  string key = 1;               // "collection" | "document_id" | "embedding_model" | "chunk_index"
  string value = 2;
}

message SearchRequest {
  string query = 1;
  string schema_name = 2;       // tenant scope — required, validated ^school_[0-9]+$ + existence (D-09)
  repeated Filter filters = 3;  // AND-combined metadata filters (D-06)
  int32 top_k = 4;              // result count; default 10, cap 100
}

message SearchResult {
  string document_id = 1;
  int32 chunk_index = 2;
  string text = 3;
  double score = 4;             // RRF-fused score (D-06, k=60)
  string citation = 5;          // "document_id#chunk_index" for the AI assistant
}

message SearchResponse {
  repeated SearchResult results = 1;
}
```
