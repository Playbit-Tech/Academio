```python
import math
import httpx
from openai import AsyncOpenAI
from tenacity import retry
from app.config import settings
from app.util.retry import embed_retry

class EmbeddingClient:
    def __init__(self) -> None:
        self._client = AsyncOpenAI(api_key=settings.AI_OPENAI_API_KEY,
                                   base_url=settings.AI_EMBEDDING_BASE_URL,
                                   timeout=settings.AI_LLM_TIMEOUT_SECONDS)
        self._dim = settings.AI_EMBEDDING_DIM
        self._batch = settings.AI_EMBEDDING_BATCH_SIZE  # <= 128 (D-05)

    @retry(**embed_retry)
    async def _embed_batch(self, texts: list[str]) -> list[list[float]]:
        resp = await self._client.embeddings.create(model=settings.AI_EMBEDDING_MODEL, input=texts)
        # order preserved by the API; index by position
        return [e.embedding for e in resp.data]

    async def embed_texts(self, texts: list[str]) -> list[list[float]]:
        if not texts:
            return []
        if any(len(t) > 8000 for t in texts):
            raise ValueError("embedding input exceeds 8000 chars")   # DoS bound
        vectors: list[list[float]] = []
        for i in range(0, len(texts), self._batch):
            vectors.extend(await self._embed_batch(texts[i:i + self._batch]))
        for v in vectors:
            if len(v) != self._dim:
                raise ValueError(f"embedding dimension {len(v)} != locked {self._dim} (PGV-04a)")
            norm = math.sqrt(sum(x * x for x in v))
            if math.isclose(norm, 0.0, abs_tol=1e-12):
                raise ValueError("zero-norm embedding rejected (cosine NaN risk)")  # RESEARCH Pitfall 2
        return vectors
```
(Batch ≤128, dimension assert fail-loud, zero-vector 400-class rejection, 8000-char input cap, tenacity retry via `embed_retry`.)

    # X-School-Schema header is ACCEPTED but not required here — /v1/embed performs
    # no DB access; the header is honored for symmetry with tenant-scoped routes (D-09).

class EmbedResponseOut(BaseModel):
    model: str
    dimension: int
    embeddings: list[list[float]]

@router.post("/embed")
async def embed(req: EmbedRequestIn) -> EmbedResponseOut:
    try:
        vecs = await EmbeddingClient().embed_texts(req.texts)
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e)) from e
    if not vecs:  # no embedding failed the dim/norm gate
        raise HTTPException(status_code=500, detail="embedding pipeline returned no vectors")
    return EmbedResponseOut(model="text-embedding-3-small", dimension=len(vecs[0]), embeddings=vecs)

from fastapi import APIRouter, Depends
from app.main import require_token
from app.providers.healthcheck import ProviderHealth
from app.providers.registry import build_provider_registry

router = APIRouter(prefix="/v1", dependencies=[Depends(require_token)])

_health = ProviderHealth()   # module-level singleton: TTL cache + cooldown state (D-10)

@router.get("/providers")
async def providers() -> dict:
    reg = build_provider_registry()
    results = []
    for name, info in reg.items():
        status = await _health.check(name, info)
        results.append({
            "provider": name,
            "status": status.status,
            "latency_ms": status.latency_ms,
            "last_checked": status.last_checked,
            "cooldown_until": status.cooldown_until,
        })
    return {"providers": results}
