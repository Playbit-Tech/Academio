```python
import functools, re
from pathlib import Path
import yaml  # PyYAML — add via uv if not present (check pyproject; if absent: `uv add pyyaml>=6`)
from jinja2 import Environment, StrictUndefined, TemplateError
from app.config import settings

# dev/staging/prod alias -> version selector (D-08). Configurable, defaults:
#   dev    -> "working" (current working tree file — always available)
#   staging-> "latest"  (highest version field among prompt.yaml files)
#   prod   -> "latest"  (same selector; Git tag is the audit trail, not runtime state)
ALIAS_DEFAULTS = {"dev": "working", "staging": "latest", "prod": "latest"}
_SUPPORTED = {"report-comments", "lesson-plans", "questions", "rubrics", "behaviour-summary",
              "attendance-analysis", "parent-letters", "meeting-minutes", "translation"}

class PromptLibrary:
    def __init__(self, prompts_dir: str | None = None) -> None:
        self._dir = Path(prompts_dir or settings.AI_PROMPTS_DIR)
        self._env = Environment(undefined=StrictUndefined, autoescape=False)  # LLM prompts: no HTML escaping; StrictUndefined catches typos
        self._cache: dict[tuple[str, str], dict] = {}

    def _load(self, prompt_type: str) -> dict:
        if prompt_type not in _SUPPORTED:
            raise ValueError(f"unknown prompt type: {prompt_type}")
        base = self._dir / prompt_type
        meta = yaml.safe_load((base / "prompt.yaml").read_text(encoding="utf-8"))
        tpl = (base / "template.txt").read_text(encoding="utf-8")
        return {"meta": meta, "template": tpl}

    def _resolve_alias(self, alias: str) -> str:
        return ALIAS_DEFAULTS.get(alias, alias)   # unknown alias treated as raw version selector

    @functools.lru_cache(maxsize=128)
    def get_prompt(self, prompt_type: str, alias: str = "prod") -> dict:
        """Cached load: (type, alias) -> {meta, template}. Git is the version mechanism (D-08)."""
        resolved = self._resolve_alias(alias)
        data = self._load(prompt_type)
        # "latest" (staging/prod default): pick the highest version among prompt.yaml files;
        # with a single committed template this equals the working tree — but the SELECTOR
        # logic must exist and be testable (D-08 alias semantics).
        return {"type": prompt_type, "alias": resolved, **data}

    def render(self, prompt_type: str, variables: dict, alias: str = "prod") -> str:
        data = self.get_prompt(prompt_type, alias)
        try:
            return self._env.from_string(data["template"]).render(**variables)
        except TemplateError as e:
            raise ValueError(f"prompt render failed for {prompt_type}: {e}") from e
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| template files → renderer | Server-authored templates (Git-tracked) cross into Jinja2 rendering |
| variables dict → renderer | Caller-supplied variables (may include user content) cross into template substitution |
| rendered prompt → provider | The composed system prompt (template + variables) crosses into the LLM call |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-07-01 | T | prompt injection via variables | mitigate | Variables are DATA substituted by Jinja2 — never evaluated/executed; templates are server-authored (Git-reviewed) and use `{{ var }}` only; tests assert zero `{%` control blocks in all nine templates; `StrictUndefined` fails render on typos instead of leaking placeholder text |
| T-03-07-02 | T | template tampering | mitigate | Templates live in the Git repo (versioned, reviewed); `AI_PROMPTS_DIR` is env-only (no request-derived path); path traversal via prompt_type blocked by the `_SUPPORTED` allowlist (9 exact names, no user paths) |
| T-03-07-03 | I | alias confusion | accept | dev/staging/prod resolve to version selectors in code defaults; misconfigured alias falls back to the alias itself as selector → load error surfaces loudly (no silent wrong prompt) |
| T-03-07-04 | D | unbounded rendered prompt | mitigate | Templates are small server-authored files; variable content bounded by the chat max_tokens cap (settings.AI_MAX_TOKENS) at the provider call — a huge variable bloats the request but is capped at inference |
| T-03-07-05 | I | secret leakage | mitigate | Templates contain no credentials (Git-reviewed); render errors surface as 400 with a sanitized message (no file contents, no keys) |
