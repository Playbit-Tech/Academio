<objective>
Extend the FROZEN Go EngineClient seam with `IngestDocument` (D-07) and add the small, additive Python `document_id` passthrough (D-02) that makes ingest idempotent across worker retries.

Purpose: ROADMAP criterion 4 — "no duplicate vectors after worker restarts." The Phase 2/3 idempotency (`UNIQUE(document_id, chunk_index)` + `ON CONFLICT DO NOTHING`) only works when the SAME document_id is used on retry; Python currently generates `uuid.uuid4()` internally, so a worker crash mid-ingest followed by a retry would produce a second document_id → duplicate vectors. Go will pass `ai_documents.id` as the stable document_id (D-02). This plan creates the contract the worker (04-03) and the SSE relay's error typing rely on; `Chat`/`ChatStream`/`Extract`/`Health` signatures stay byte-identical (interface-conformant extension only).
Output: `IngestDocumentRequest/Response` + `StatusError` in engine.go, `httpClient.IngestDocument` in client.go with httptest coverage, and the Python `document_id` passthrough with tests.
</objective>

```go
// SchoolSchemaHeader carries the tenant schema on tenant-scoped calls.
const SchoolSchemaHeader = "X-School-Schema"

// IngestDocumentRequest is the payload for IngestDocument (POST /v1/documents).
type IngestDocumentRequest struct {
	DocumentPath string `json:"document_path"`
	Collection   string `json:"collection"`
	DocumentID   string `json:"document_id"`
	// SchoolSchema is the tenant schema (e.g. "school_1"). JSON-skipped —
	// sent as the X-School-Schema header, never in the body (D-09).
	SchoolSchema string `json:"-"`
}

// IngestDocumentResponse mirrors Python's /v1/documents response (D-03/D-07).
type IngestDocumentResponse struct {
	Status    string   `json:"status"`
	DocumentID string  `json:"document_id"`
	Chunks    int      `json:"chunks"`
	Pages     int      `json:"pages"`
	OcrPages  int      `json:"ocr_pages"`
	Chars     int      `json:"chars"`
	Warnings  []string `json:"warnings"`
}

// StatusError carries the HTTP status of a failed engine call so callers can
// classify permanent vs transient failures (D-04) without string parsing.
type StatusError struct {
	StatusCode int
	Body       string
}

func (e *StatusError) Error() string {
	return fmt.Sprintf("engine: unexpected status %d: %s", e.StatusCode, e.Body)
}
```

In `client.go` implement:
```go
func (c *httpClient) IngestDocument(ctx context.Context, req IngestDocumentRequest) (*IngestDocumentResponse, error) {
	ctx, cancel := context.WithTimeout(ctx, extractTimeout)
	defer cancel()
	body, err := json.Marshal(req)
	if err != nil {
		return nil, fmt.Errorf("engine ingest marshal: %w", err)
	}
	hreq, err := c.newRequest(ctx, http.MethodPost, "/v1/documents", body)
	if err != nil {
		return nil, err
	}
	if req.SchoolSchema != "" {
		hreq.Header.Set(SchoolSchemaHeader, req.SchoolSchema) // D-09: tenant-scoped, no fallback
	}
	resp, err := c.client.Do(hreq)
	if err != nil {
		return nil, fmt.Errorf("engine ingest: %w", err)
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		respBody, err := io.ReadAll(resp.Body) // Rule B1: NEVER discard errors with _
		if err != nil {
			return nil, fmt.Errorf("engine ingest read body: %w", err)
		}
		return nil, &StatusError{StatusCode: resp.StatusCode, Body: string(respBody)}
	}
	var out IngestDocumentResponse
	if err := json.NewDecoder(resp.Body).Decode(&out); err != nil {
		return nil, fmt.Errorf("engine ingest decode: %w", err)
	}
	return &out, nil
}
```

```python
async def ingest_document(path: str, schema_name: str, collection: str = "default",
                          document_id: str | None = None) -> dict:
    extraction = await extract_document(path)
    chunks = chunk_text(extraction.text)
    if not chunks:
        return {
            "status": "success",
            "document_id": document_id,   # D-02: echo caller id even with 0 chunks
            "chunks": 0,
            "pages": extraction.pages,
            "ocr_pages": extraction.ocr_pages,
            "warnings": ["no text extracted"],
        }
    vectors = await EmbeddingClient().embed_texts(chunks)
    doc_id = document_id or str(uuid.uuid4())   # D-02: caller wins, uuid4 fallback
    inserted = await insert_chunks(schema_name, collection, doc_id, [...], settings.AI_EMBEDDING_MODEL)
    return {"status": "success", "document_id": doc_id, "chunks": inserted, ...}
```
