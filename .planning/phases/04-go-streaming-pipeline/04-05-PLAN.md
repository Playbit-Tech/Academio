<objective>
Implement the SSE relay endpoint `GET /api/v2/ai/chat/stream` (D-05) on the AIHandler: stream engine deltas/citations/usages to the browser over a shared event envelope with heartbeats, bounded buffering, and upstream cancellation on disconnect.

Purpose: INT-01 — real-time token streaming to the frontend. The engine client already returns an SSE stream via `EngineClient.ChatStream` (04-01 seam, `StreamCallback func(EngineEvent) error`, context-bound). This plan bridges that to the browser: the relay drains the engine stream through a bounded channel (cap 64), selects between forwarding and aborting when a slow client can't keep up, sends `: ping` heartbeats so proxies/EventSource don't time out, sets `X-Accel-Buffering: no` + `Cache-Control: no-cache`, and translates engine events into the single shared envelope (`delta | citation | usage | error | done`) — in-band errors after HTTP 200 included (ROADMAP criterion 2). The engine seam itself is NOT modified in behavior: `ChatStream`'s context-bound semantics already satisfy upstream cancellation when the relay's goroutine returns.
Output: `AIHandler.StreamChat` + `stream.go` relay + router registration + unit tests (D-10). Wired in setup.go with the existing AIHandler + engine client.
</objective>

```go
package ai

import "time"

// Event names in the shared envelope (keep EXACT — frontend contract, D-05).
const (
    sseEventDelta    = "delta"
    sseEventCitation = "citation"
    sseEventUsage    = "usage"
    sseEventError    = "error"
    sseEventDone     = "done"
)

// Package vars (NOT consts) so tests can shrink them (Task 3).
var (
    relayBufferSize = 64                  // bounded channel cap (D-05)
    heartbeatEvery  = 25 * time.Second    // <= 30s (D-05)
)

func (h *AIHandler) StreamChat(c *gin.Context) {
    // 1. Request parse — bind the MODULES/ai request DTO (Message, ConversationID,
    //    AgentType, Model, Temperature — it has NO Stream field; D-05)
    var req ChatRequest
    if err := c.ShouldBindJSON(&req); err != nil { response.Error(c, 400, "invalid_chat_request", "invalid request body", "validation"); return }
    if req.Message == "" { response.Error(c, 400, "invalid_chat_request", "message is required", "validation"); return }

    // 2. Map to the ENGINE seam's request shape (engine.ChatRequest, NOT modules/ai.ChatRequest):
    //    {Model string, Messages []engine.ChatMessage, Stream bool} — see <interfaces> below.
    engineReq := engine.ChatRequest{
        Model:    req.Model, // may be empty → engine/agent default
        Messages: []engine.ChatMessage{{Role: "user", Content: req.Message}},
        Stream:   true,      // always streaming on this route (INT-01)
    }

    // 3. SSE headers BEFORE any write (nginx-safe, D-05)
    c.Header("Content-Type", "text/event-stream")
    c.Header("Cache-Control", "no-cache")
    c.Header("X-Accel-Buffering", "no")
    c.Header("Connection", "keep-alive")
    c.Writer.Flush() // ensure headers hit the wire immediately

    // 4. Bounded relay channel + upstream ctx
    events := make(chan engine.EngineEvent, relayBufferSize)
    ctx, cancel := context.WithCancel(c.Request.Context()) // disconnect propagates upstream
    defer cancel()

    // 5. Drainer goroutine: engine -> channel (context-bound; no rescan of SSE here)
    go func() {
        defer close(events)
         cb := func(ev engine.EngineEvent) error {
             if ev.Type == sseEventDone { // Python emits done_event (chat.py:197); relay synthesizes ONE terminal done below (dedup contract)
                 return nil
             }
             select {
            case events <- ev:
                return nil
            default: // buffer full = slow client (D-05)
                return fmt.Errorf("relay buffer full: aborting upstream")
            }
        }
        if err := h.aiClient.ChatStream(ctx, engineReq, cb); err != nil {
            // emit terminal in-band error (ROADMAP criterion 2), best-effort
            ev, merr := json.Marshal(map[string]string{"message": err.Error()})
            if merr != nil { // Rule B1: never discard
                logger.Errorf("stream error event marshal", "error", merr)
                return
            }
            select {
            case events <- engine.EngineEvent{Type: sseEventError, Data: ev}:
            default: // already full — skip; client is gone
            }
        }
    }()

    // 6. Heartbeat ticker (<=30s, D-05)
    ticker := time.NewTicker(heartbeatEvery)
    defer ticker.Stop()

    // 7. Forward loop — SSE frames MUST be `data: {json}\n\n` (envelope contract,
    //    Task 3 asserts them); never raw JSON via json.NewEncoder(c.Writer).
    for {
        select {
        case <-c.Request.Context().Done(): // client gone
            return
        case <-ticker.C:
            c.Writer.WriteString(": ping\n\n") // comment frame — never breaks parsing (D-05)
            c.Writer.Flush()
        case ev, ok := <-events:
            if !ok {
                done, merr := json.Marshal(map[string]interface{}{"event": sseEventDone, "data": map[string]interface{}{}})
                if merr != nil { return } // static payload; guard keeps Rule B1
                if _, err := fmt.Fprintf(c.Writer, "data: %s\n\n", done); err != nil {
                    return // client gone
                }
                c.Writer.Flush()
                return
            }
            payload, err := json.Marshal(map[string]interface{}{"event": ev.Type, "data": json.RawMessage(ev.Data)})
            if err != nil { logger.Errorf("stream marshal", "error", err); return }
            if _, err := fmt.Fprintf(c.Writer, "data: %s\n\n", payload); err != nil { return } // write error = disconnect
            c.Writer.Flush() // real-time: flush every event
        }
    }
}
```
